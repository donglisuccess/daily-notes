## 一、接口状态码
```py
from enum from Enum

class HttpCode(str, Enum):
    SUCCESS = "success" # 成功
    FAIL = "fail" # 失败
    VALIDATE_ERROR = "validate_error" # 数据校验错误
    UNAUTHORIZED = "unauthorized" # 没有登录信息
    FORBIDDEN = "forbidden" # 登录了，但是没有权限
```

## 二、接口响应格式
```py
from .http_code import HttpCode
from typing import Any
from dataclasses import dataclass

@dataclass
class Response:
    code: HttpCode = HttpCode.SUCCESS
    data: Any = {}
    message: str = ""
```

## 三、封装具体状态下的响应
```py
from flask import jsonify

class json:
    def __init__(self, data)
    return jsonify(data), 200 # 规定所有的响应必须为200
```
**封装方法**
```py
def success_json(data):
    return json({code: HttpCode.SUCCESS, data: data, message: ""})

def fail_json(data):
    return json({code: HttpCode.FAIL, data: data, message: ""})

def validate_json(data):
    first_key = next(iter(data))
    if first_key:
        msg = data.get(first_key)[0]
    else:
        msg = ""
    return json({code: HttpCode.VALIDATE_ERROR, data: data, message: ""})
```
**封装消息的方法**
```py
def message(code:HttpCode, message: str):
    return json({code: code, message: message, data: {}})

def success_message(message: str):
    return message(HttpCode.SUCCESS, message: message)

def fail_message(message: str):
    return message(HttpCode.FAIL, message: message)

def unauthorized_message(message: str):
    return message(code:HttpCode.UNAUTHORIZED, message: message) 

def forbidden_message(message: str):
    return message(code: HttpCode.FORBIDDEN, message: message)
```
