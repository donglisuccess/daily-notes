## 一、封装错误报错
在 `internal.exception` 文件夹中编写错误封装处理类。
```py
from pkg.response import HttpCode
from typing import Any

class CustomException(Exception):
    code:HttpCode = HttpCode.FAIL
    data:Any = None
    message:str = None

    def __init__(self, data:Any = None, message:str = None):
        self.data = data
        self.message = message

class FailException(CustomException):
    pass

# not-found消息体
class NotFoundException(CustomException):
    code = HttpCode.NOT_FOUND
    pass

# 没有授权消息体
class UnauthorizedException(CustomException):
    code = HttpCode.UNAUTHORIZED
    pass

# 没有权限消息体
class ForbiddenException(CustomException):
    code = HttpCode.FORBIDDEN
    pass
```
统一导出：
```py
from .exception import (
    CustomException,
    FailException,
    NotFoundException,
    UnauthorizedException,
    ForbiddenException
)

__all__ = ["CustomException", "FailException", 
    "NotFoundException", "UnauthorizedException", 
    "ForbiddenException"]
```

## 二、在服务配置中捕获
```py
from pkg.response import json, Response, HttpCode
import os

class HTTP(Flask):
    def __init__(self):
        ...
        ...
        # 处理报错消息
        self.register_error_handler(Exception, self._register_error_handler)

    def _register_error_handler(self, error: Exception):
        # 1、如果是我们自定义报错，处理如下：
        if isinstance(error, CustomException):
            # 直接返回状态码以及消息即可。
            return json(Response(
                code=error.code
                data=error.data
                message=error.message
            ))
        else:
        # 2、如果非我们自己定义的报错，处理如下：
            # i、如果是开发环境或者服务的debug为true，则直接报错
            if os.getenv("FLASK_ENV") == "development" or self.debug == True:
                raise error
            else:
                return json(Response(
                    code=HttpCode.FAIL
                    message=error.message
                    data=error.data
                ))
```
