## 一、摘要缓存记忆实现
```py
#!/usr/bin/env python
# -*- coding: utf-8 -*-
# @Time    : 2026/11/15 17:56
# @Author  : LENOVO
# @File    : 缓存摘要记忆.py
# @Software: PyCharm
# @Project : llmops-api

"""
缓存摘要记忆
"""
import os
from openai import OpenAI
from dotenv import load_dotenv
load_dotenv()

client = OpenAI(
    api_key=os.getenv("DEEPSEEK_API_KEY"),
    base_url=os.getenv("DEEPSEEK_BASE_URL"),
)

class ConversationMomerySummary:
    """摘要总结类"""
    def __init__(self, chat_histroy: list, chat_summery: str, max_token: int):
        # 历史会话记录
        self.chat_histroy = [] if chat_histroy is None else chat_histroy
        # 历史消息总结
        self.chat_summery = chat_summery
        # 最长的token数
        self.max_token = max_token
        # 用作总结的client
        self._client = OpenAI(
            api_key=os.getenv("DEEPSEEK_API_KEY"),
            base_url=os.getenv("DEEPSEEK_BASE_URL")
        )

    # 获取当前的历史消息长度
    def get_chat_histroy_tokens_num(self) -> int:
        return len(self.history_messages_to_string())

    # 将历史消息列表转化为字符串
    def history_messages_to_string(self) -> str:
        history_str = ""
        for message in self.chat_histroy:
            history_str += f"human:{message["human"]} ai: {message["ai"]}\n"
        return history_str


    # 将新的消息push到历史消息列表中
    def push_message_to_history(self, message: dict):
        self.chat_histroy.append(message)

        # 判断大小，暂定先取出第一条数据和历史摘要做压缩
        if self.get_chat_histroy_tokens_num() >= self.max_token:
            self.summarize_history_with_summary()

    # 将历史消息和摘要总结成新的摘要
    def summarize_history_with_summary(self):
        first_message = self.chat_histroy[0]
        print("开始对历史会话进行压缩处理~")
        response = self._client.chat.completions.create(
            model="deepseek-flash",
            messages=[
                {"role": "system",
                 "content": "你是一个专业做历史消息压缩的AI应用，现在针对于我的历史消息和摘要做压缩，然后给我一个新的摘要"},
                {"role": "user",
                 "content": f"human:{first_message['human']} ai: {first_message['ai']} summery：{self.chat_summery}"}
            ]
        )
        self.chat_summery = response.choices[0].message.content
        del self.chat_histroy[0]
        print(f"对历史会话压缩成功，新摘要为:{self.chat_summery}")

mermory = ConversationMomerySummary([], "", 300)
while True:
    query = input("👨:")
    prompt = f"""
        你是一个有用的AI应用，可以帮助用户回答问题。请你基于现有摘要和历史消息回答用户问题：
        
        摘要：{mermory.chat_summery}
        
        历史消息：{mermory.history_messages_to_string()}
    """

    response = client.chat.completions.create(
        model="deepseek-flash",
        messages=[
            {"role": "system", "content": prompt},
            {"role": "user", "content": query}
        ],
        stream=True
    )
    ai_content = ""
    print("🤖:", end="")
    for chunk in response:
        if chunk.choices[0] and chunk.choices[0].delta.content:
            content = chunk.choices[0].delta.content
            ai_content += content
            print(content, flush=True, end="")
    mermory.push_message_to_history({
        "human": query,
        "ai": ai_content
    })
    print("\n")
```
